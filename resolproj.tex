\section{The polygraphic resolution theorem}

\subsection{Folk model structure}
\begin{paragr}
  Recall that there exists a so-called ``folk'' model category
  structure on the category $\oCat$ \cite{lafontetal:folkms}, characterized by the fact that
  all objects are fibrant and the cofibrations are generated by the
  set 
  \[
    \setof{\gcof{n}:\Sph {n-1} \to \Dsk n }{n \in \N}.
  \]
  The cofibrant objects of this model structure are exactly the
  \oo\nbd-categories free on a polygraph. For a description of the
  weak equivalences see \cite[Theorem 6.19]{lafontetal:folkms}. 
\end{paragr}
\begin{paragr}
  More generally, for every $k \in \N$, there exists a model
  category structure on the category $\opCat{k}$ \cite{ara_lucas:folmon} which is
  right-induced from the inclusion functor
  \[
    \oemb k:\opCat{k} \to \oCat.
  \]
  In other words, every object of $\opCat{k}$ is fibrant and, if
  $\opifun{k} \colon \oCat \to \opCat{k}$ is the left adjoint of the
  previous inclusion functor, which formally inverts all cells of
  dimension $> k $, then the cofibrations of $\opCat{k}$ are generated
  by the set 
  \[
    \setof{\opifun{k}\gcof{n}:\opifun{k}\Sph {n-1} \to \opifun{k}\Dsk n }{n \in \N}.
  \]
\end{paragr}

\begin{paragr}
  As with any cofibrantly generated model structure, given a small
  category $C$, we can consider the \emph{projective} model structure
  on the category of functors
  \[
    \fun{C}{\opCat{k}},
  \]
  for any $k \in \N\cup\{\omega\}$. The weak equivalences, fibrations
  and trivial fibrations are the pointwise ones (in particular, every
  object is fibrant) and the cofibrations are generated by the set of
  morphisms
  \[
   \coprod_{\homset{C}{c}{-}}\opifun{k}\gcof{n}: \coprod_{\homset{C}{c}{-}}\opifun{k}\Sph{n-1} \to \coprod_{\homset{C}{c}{-}}\opifun{k}\Dsk{n},
  \]
  for all object $c$ of $C$ and $n \in \N$.
\end{paragr}
\subsection{The polygraphic resolution theorem}\label{subsection:polresol}
\begin{paragr}
  Let $C$ be a small category. Recall that each object $c$ of $C$
  determines the slice category $\tr Cc$ whose objects are pairs
  $\pair{x}{f:x\to c}$ where $x$ is an object of $C$ and $f:x\to c$ is
  a morphism in $C$. A morphism $\pair xf\to \pair{x'}{f'}$ is a
  morphism $g:x\to x'$ in $C$ such that $f'g=f$. The correspondence
  $c\mapsto \tr Cc$ is then the object part of a functor
\[
    \begin{aligned}
      \tr{C}{-} \colon C &\mapsto \Cat \\
      c &\mapsto \tr{C}{c}.
    \end{aligned}
  \]
  Also, for each object $c$, we get a canonical forgetful functor
  \[\tr Cc\to C\]
  taking $\pair x{f:x\to c}$ to $x$.
\end{paragr}

\begin{lemma}\label{lemma:slicescond}
  For each small category $C$ the canonical functor
 \[\tr Cc\to C\]
  is a discrete Conduché fibration.
\end{lemma}
\begin{proof}
Consider a morphism in $\tr Cc$ from $\pair xf$ to $\pair x{f'}$, that
is, a morphism $g:x\to x'$ in $C$ such that the following triangle commutes:
\[
    \begin{tikzcd}[column sep=small]
      x \ar[rr,"g"] \ar[dr,"f"'] && x'\ar[dl,"f'"] \\
      &c&
    \end{tikzcd}.
  \]
  If $g=g_2\circ g_1$, then we have the following decomposition in
  $\tr{C}{c}$,
  \[
      \begin{tikzcd}
      x \ar[rr,"g"] \ar[dr,"f"'] && x'\ar[dl,"f'"] \\
      &c&
    \end{tikzcd}
    =
    \begin{tikzcd}
      x \ar[r,"g_1"] \ar[dr,"f"']& y \ar[r,"g_2"] \ar[d,"h"] &
      x' \ar[dl,"f'"] \\
      &c&
    \end{tikzcd},
  \]
  with $h=f'\circ g_2'$. This choice of $h$ is clearly unique, thus
  proving that $\tr{C}{x} \to C$ is a discrete Conduché fibration.
\end{proof}

\begin{theorem}\label{thm:projresol}
  Let $C$ be a (small) $\pk 1 k$\nbd-category with $k \in \{0,1\}$ (i.e.\
  $C$ is a category or a groupoid),  $i \in \N\cup\{\omega\}$ such that
  $i \geq k$, and $P \to C$ be a polygraphic
  resolution in $\opCat{i}$. The functor
    \[
    \begin{aligned}
      \tr{P}{-} \colon C &\to  \opCat{i}\\
      c &\mapsto \tr{P}{c}
    \end{aligned}
  \]
  is a cofibrant object of the category
  $\fun{C}{\opCat{i}}$, equipped with the projective model structure.
\end{theorem}


The first ingredient to prove this theorem is the following lemma.
\begin{proof}[Proof of \cref{thm:projresol}]
By \cref{lemma:slicescond} and the fact that the class of discrete Conduché fibrations is
stable by pullback, it follows that for every object $c$ of $C$, the
morphism
\[
  \tr{P}{c} \to P
\]
is a discrete Conduché fibration. By \cref{thm:main}, it follows that
$\tr{P}{c}$ is a polygraph in $\opCat{i}$, and more precisely by \cref{prop:sum_up}, if
$\Sigma_n$ is a set of generators of dimension $n$ of $P$, then the
set
\[
  \tr{\Sigma_n}{c}:=\setof{(x,f) \in (\tr{P}{c})_n}{x \in \Sigma_n}
\]
is a set of generators of dimension $n$ of $\tr{P}{c}$. Notice
that, by definition, we have
\[
  \tr{\Sigma_n}{c} = \coprod_{x \in \Sigma_n}\homset{C}{\target_0(x)}{c}.
\]
Now, $\tr{\Sigma_n}{c}$ being a set of generators of dimension $n$ of
$\tr{P}{c}$, we have a pushout square
\[
  \begin{tikzcd}
   \displaystyle \coprod_{x \in
      \Sigma_n}\coprod_{f \in \homset{C}{\target_0(x)}{c}}\ar[d]\opifun{i}\Sph{n-1}\ar[r]&
    \trk{n-1}(\tr{P}{c})\ar[d] \\
    \displaystyle \coprod_{x \in
      \Sigma_n}\coprod_{f\in\homset{C}{\target_0(x)}{c}}\opifun{i}\Dsk{n}\ar[r]&\trk{n}(\tr{P}{c})
    \ar[from=1-1,to=2-2,phantom,very near end,"\ulcorner"]
  \end{tikzcd}
\]

  troncations.}
All the morphisms in the above square are natural in $c$, and thus
this defines a pushout square in $\fun{C}{\opCat{i}}$
\[
  \begin{tikzcd}
   \displaystyle \coprod_{x \in
      \Sigma_n}\coprod_{f \in \homset{C}{\target_0(x)}{-}}\ar[d]\opifun{i}\Sph{n-1}\ar[r]&
    \trk{n-1}(\tr{P}{-})\ar[d] \\
    \displaystyle \coprod_{x \in
      \Sigma_n}\coprod_{f\in\homset{C}{\target_0(x)}{-}}\opifun{i}\Dsk{n}\ar[r]&\trk{n}(\tr{P}{-}).
    \ar[from=1-1,to=2-2,phantom,very near end,"\ulcorner"]
  \end{tikzcd}
\]
By stability of cofibrations by coproducts and pushout, this proves
that \[\trk{n-1}(\tr{P}{-})\to\trk{n}(\tr{P}{-})\] is a cofibration for
the projective model structure on $\fun{C}{\opCat{i}}$. Note in
particular that for $n=0$, this just means that $\trk{0}(\tr{P}{-})$
is cofibration. Now, since $\tr{P}{-}$ is the colimit of
\[
  \trk{0}(\tr{P}{-}) \rightarrow \trk{1}(\tr{P}{-}) \rightarrow \cdots,
\]
this proves that $\tr{P}{-}$ is cofibrant.
\end{proof}
