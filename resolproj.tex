\section{The polygraphic resolution theorem}

\subsection{Folk model structure}
\begin{paragr}\label{paragr:folk}
  Recall that there exists a so-called ``folk'' model category
  structure on the category $\oCat$ \cite{lafontetal:folkms}, characterized by the fact that
  all objects are fibrant and the cofibrations are generated by the
  set 
  \[
    \setof{\gcof{n}:\Sph {n-1} \to \Dsk n }{n \in \N}.
  \]
  The cofibrant objects of this model structure are exactly the
  \oo\nbd-categories free on a polygraph. For a description of the
  weak equivalences see \cite[Theorem 6.19]{lafontetal:folkms}. 
\end{paragr}
\begin{paragr}
  More generally, for every $k \in \N$, there exists a model
  category structure on the category $\opCat{k}$ \cite{ara_lucas:folmon} which is
  right-induced from the inclusion functor
  \[
    \oemb k:\opCat{k} \to \oCat.
  \]
  In other words, every object of $\opCat{k}$ is fibrant and the cofibrations of $\opCat{k}$ are generated
  by the set 
  \[
    \setof{\opifun{k}\gcof{n}:\opifun{k}\Sph {n-1} \to \opifun{k}\Dsk n }{n \in \N},
  \]
  where $\opifun{k} \colon \oCat \to \opCat{k}$ is left adjoint to
  $\oemb k$ (see~\cref{paragr:invert}).
\end{paragr}
\begin{definition}\label{def:respol}
  Let $k\in \N\cup\set{\oo}$ and $C$ an $\ook k$-category. A {\em
  polygraphic resolution} of $C$ in $\opCat k$ consists of an
$\ook k$-polygraph $P$ together with an acyclic fibration
$p:\frgp P\to C$
in the folk model structure of $\opCat k$.
\end{definition}
\begin{paragr}
  As with any cofibrantly generated model structure, given a small
  category $C$, we can consider the \emph{projective} model structure
  on the category of functors
  \[
    \fun{C}{\opCat{k}},
  \]
  for any $k \in \N\cup\{\omega\}$. The weak equivalences, fibrations
  and trivial fibrations are the pointwise ones (in particular, every
  object is fibrant) and the cofibrations are generated by the set of
  morphisms
  \[
   \coprod_{\homset{C}{c}{-}}\opifun{k}\gcof{n}: \coprod_{\homset{C}{c}{-}}\opifun{k}\Sph{n-1} \to \coprod_{\homset{C}{c}{-}}\opifun{k}\Dsk{n},
  \]
  for all object $c$ of $C$ and $n \in \N$ (see~\cite[Theorem 11.6.1]{hirschhorn:modctl}).
\end{paragr}
\subsection{Slice categories}\label{subsection:slice_cat}
\begin{paragr}
  Let $C$ be a small category. Recall that each object $c$ of $C$
  determines the slice category $\tr Cc$ whose objects are pairs
  $\pair{x}{f:x\to c}$ where $x$ is an object of $C$ and $f:x\to c$ is
  a morphism in $C$. A morphism $\pair xf\to \pair{x'}{f'}$ is a
  morphism $g:x\to x'$ in $C$ such that $f'g=f$. The correspondence
  $c\mapsto \tr Cc$ is then the object part of a functor
\[
    \begin{aligned}
      \tr{C}{-} \colon C &\mapsto \Cat \\
      c &\mapsto \tr{C}{c}.
    \end{aligned}
  \]
  Also, for each object $c$, we get a canonical forgetful functor
  $\sld:\tr Cc\to C$
  taking $\pair x{f:x\to c}$ to $x$.
\end{paragr}

\begin{lemma}\label{lemma:slicescond}
  For each small category $C$ the canonical functor
 $\sld:\tr Cc\to C$
  is a discrete Conduché fibration.
\end{lemma}
\begin{proof}
Consider a morphism in $\tr Cc$ from $\pair xf$ to $\pair{x'}{f'}$, that
is, a morphism $g:x\to x'$ in $C$ such that the following triangle commutes:
\[
    \begin{tikzcd}[column sep=small]
      x \ar[rr,"g"] \ar[dr,"f"'] && x'\ar[dl,"f'"] \\
      &c&
    \end{tikzcd}.
  \]
  If $g=g_2\circ g_1$, then we have the following decomposition in
  $\tr{C}{c}$,
  \[
      \begin{tikzcd}
      x \ar[rr,"g"] \ar[dr,"f"'] && x'\ar[dl,"f'"] \\
      &c&
    \end{tikzcd}
    =
    \begin{tikzcd}
      x \ar[r,"g_1"] \ar[dr,"f"']& y \ar[r,"g_2"] \ar[d,"h"] &
      x' \ar[dl,"f'"] \\
      &c&
    \end{tikzcd},
  \]
  with $h=f'\circ g_2'$. This choice of $h$ is clearly unique, thus
  proving that $\tr{C}{x} \to C$ is a discrete Conduché fibration.
\end{proof}

\subsection{Polygraphic resolutions of small categories}\label{subsec:polres}
\begin{paragr}\label{paragr:pullback_resol}
From now on, we shall identify any small category with its image
in $\opCat 1$ by the inclusion functor $\oinc 1:\npCat 11\to \opCat 1$
(\ref{ssubsec:ncat}), and all diagrams are drawn in $\opCat 1$. Thus
let $C$ be a small category, $P$ an \ook 1-polygraph and $p:\frgp P\to
C$ a polygraphic resolution of $C$ in $\opCat 1$. Given an object $c$
of $C$, we may form the pullback of $\sld:\tr Cc\to C$ by $p:\frgp
P\to C$ in $\opCat 1$:
  \begin{equation}
  \label{eq:pullback}
  \begin{tikzcd}
      \tr{\frgp P\!}{c} \ar[d] \ar[r] & \tr{C}{c} \ar[d,"\sld"] \\
      \frgp P \ar[r,"p"'] & C.
      \ar[from=1-1,to=2-2,phantom,"\lrcorner",very near start]
    \end{tikzcd}
  \end{equation}
  Beware that $\tr{\frgp P\!}{c}$ is a mere notation for the resulting
  \ook 1-category, as we did not define slices of \ook
  1-categories. Let us describe the $n$-cells of $\tr{\frgp P\!}{c}$
  for all $n\geq 0$:
  \begin{itemize}
  \item a $0$-cell of $\tr{\frgp P\!}{c}$ is a pair $\pair{w_0}{f}$
    where $w_0\in (\frgp P)_0$, $f:x\to c\in (\tr Cc)_0$ and
    $p_0w_0=x$;
  \item for $n>0$, an $n$-cell of $\tr{\frgp P\!}{c}$ amounts to a
    pair $\pair{w_n}{f}$ where  $w_n\in (\frgp P)_n$ and $f:\tge 0
    (p_nw_n)\to c$.
  \end{itemize}
Clearly, given a resolution $p$, the construction of $\tr{\frgp
  P\!}{c}$ is functorial in $c$, thus defining a functor
  \[
    \begin{aligned}
      \tr{\frgp P\!}{-} \colon C &\to \opCat{k} \\
      c &\mapsto \tr{\frgp P\!}{c}.
    \end{aligned}
  \]
\end{paragr}
\begin{theorem}\label{thm:projresol}
    Let $C$ be a small category and $p:\frgp P\to C$ a polygraphic
    resolution of $C$ in $\opCat 1$. The functor
    \[\tr{\frgp P\!}{-}:C\to \opCat 1\]
    is a cofibrant object of the category $\fun{C}{\opCat{1}}$, equipped with the projective model structure.
\end{theorem}
  \begin{proof}
   Let $c$ be an object of $C$ and consider the
   square~(\ref{eq:pullback}). By~\cref{lemma:slicescond}, the
   morphism $\sld:\tr Cc\to C$ is a discrete Conduché fibration. Therefore
   the pullback morphism $\tr{\frgp P\!}c\to \frgp
   P$ is also a discrete Conduché fibration. As $P$ is an $\ook
   1$-polygraph, it follows from \cref{thm:main} and the description
   of the cells of $\tr{\frgp P\!}c$
   that $\tr{\frgp P\!}c$ is itself freely generated by a
   polygraph $\tr Pc$ whose set of $n$-generators is
   \[
     (\tr Pc)_n=\setof{\pair{a}{f}}{a\in P_n\ \hbox{and}\  f:\tge 0(p_n\gni n a)\to c}.
   \]
   where $\gni n$ inserts the $n$-generators in the $n$-cells of
   $\frgp P$ (see~\ref{paragr:okpol}). In other words we can say that
   $\tr{\frgp{P}\!}c\simeq \frgp{(\tr Pc)}$.

   The pushout square~(\ref{eq:polpushout})
   of~\ref{paragr:opolpushout} applies to the \ook 1-polygraph $\tr
   Pc$ and yields a pushout
   \[
    \xymatrix{
     \coprod_{{(\tr Pc)}_{n}} \opifun 1 \Sph{n-1}
     \ar[d]_{\coprod_{{(\tr Pc)}_{n}}\opifun 1\gcof {n}}\ar[r]& \oinc{n-1}\trk{n-1}\frgp {(\tr Pc)}\ar[d]\\
      \coprod_{{(\tr Pc)}_{n}}\opifun 1\Dsk{n}\ar[r]& \oinc{n}\trk {n}\frgp {(\tr Pc)}
    }
  \]
  in $\opCat 1$ for all $n\geq 1$. All the morphisms in the above
  square are natural in $c$, so that we get corresponding pushout
  squares in~$\fun{C}{\opCat{1}}$:
   \[
    \xymatrix{
     \coprod_{{(\tr P-)}_{n}} \opifun 1 \Sph{n-1}
     \ar[d]_{\coprod_{{(\tr P-)}_{n}}\opifun 1\gcof {n}}\ar[r]& \oinc{n-1}\trk{n-1}(\tr{\frgp{P}\!}{-})\ar[d]\\
      \coprod_{{(\tr P-)}_{n}}\opifun 1\Dsk{n}\ar[r]& \oinc{n}\trk {n}(\tr{\frgp{P}\!}{-})
    }
  \]
  Using the above description of the generators of $\tr P-$, the left
  map decomposes as
  \[
    \coprod_{{(\tr P-)}_{n}}\opifun 1\gcof {n}=\coprod_{a\in
      P_n}\coprod_{f\in\homset{C}{\tge 0(p_n\gni na)}{-}}\opifun 1\gcof {n}
  \]
  which is a coproduct of generating cofibrations
  in~$\fun{C}{\opCat{1}}$ by~\ref{paragr:folk}, hence a cofibration. As cofibrations are
  stable by pushout, the maps
  \[
    \oinc{n-1}\trk{n-1}(\tr{\frgp{P}\!}{-})\to \oinc{n}\trk
    {n}(\tr{\frgp{P}\!}{-})
\]
   are cofibrations. Finally, the functor
  $\tr{\frgp{P}\!}{-}$ is the colimit of the sequence of cofibrations
  \[
    \xymatrix@C=0.5cm{0\ar[r] &\oinc 0\trk 0(\tr{\frgp{P}\!}{-})\ar[r] &\cdots\ar[r]
      &\oinc{n-1}\trk{n-1}(\tr{\frgp{P}\!}{-})\ar[r]&
      \oinc{n}\trk{n}(\tr{\frgp{P}\!}{-})\ar[r] &\cdots},
  \]
  where $0$ is the initial object, and therefore a cofibrant object.
  
  \end{proof}
