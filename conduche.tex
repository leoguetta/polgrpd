\section{Basis lifting theorem}\label{sec:basislift}

\subsection{Conduché fibrations}\label{subsec:conduche}
\begin{paragr}
  Let $k\geq 0$ and $C$, $D$ be \ook k-categories. A morphism $f:C\to D$ is a {\em discrete Conduché fibration} if it satisfies the following conditions:
  \begin{enumerate}
  \item For $0\leq n<m$ and  $y\in D_n$,  $x\in C_m$ such that $f(x)=\unit
    m(y)$, there is a unique $z\in C_n$ such that $x=\unit m(z)$.
    \item For $0\leq i<n$, if $y_1$, $y_2$ are $i$-composable cells in
      $D_n$ and $x\in C_n$ such that $f(x)=y_1\comp i y_2$, there is a unique pair $\pair{x_1}{x_2}$ of $i$-composable cells in $C_n$ such that $x=x_1\comp i x_2$, $f(x_1)=y_1$ and $f(x_2)=y_2$.
    \end{enumerate}
  \end{paragr}
  \begin{remark}
    In fact, condition~(2) implies condition~(1), but due to the
    design of the formal expressions of~\ref{ssubsec:formex}, it makes the
    proofs easier when treated separately.
  \end{remark}
\begin{paragr}  
In the case where $C$, $D$ are plain \oo-categories, that is if
$k=\omega$, we know from~\cite[Theorem 6.11]{guetta:poldcf} that if
$f:C\to D$ is a discrete Conduché fibration and $D$ is freely
generated by a polygraph, then so is $C$. We aim to prove that the
result still holds for any $k\in\N$.
  \end{paragr}
  \begin{remark}
     A key feature of a free \oo-category is that its generators are
     exactly the indecomposable cells, hence uniquely determined
     (see~\cite[Section 4, Proposition 8.3]{makkai:worcom}). This is
     no longer true for $\pair{\omega}{k}$-categories when $k<\omega$, which makes the general case a
     priori significantly more complicated. However, the proof
     of~\cite[Theorem 6.11]{guetta:poldcf} does not rely on the
     uniqueness of the basis, and adapts to the general case.
  \end{remark}
    
\subsection{Lifting of free
  $(\omega,k)$-categories}\label{subsec:lifting}
\begin{paragr}
  Let us fix an integer $k\in\N$ and two $\pair{\omega}{k}$-categories
$C$, $D$ such that $D$ is the free $\pair{\omega}{k}$-category
generated by an $\pair{\omega}{k}$-polygraph $Q$, that is
$D=\frgp{Q}$. We prove that if $f:C\to D$ is a discrete Conduché
fibration, then $C$ is itself freely generated by an
$\pair{\omega}{k}$-polygraph $P$. This amounts to prove that, for each
$n\in \N$, the $\pair nk$-category $\trk n(C)$ is freely generated by
a $\pair nk$-polygraph. We reason by induction on $n$. Note that for
$n\leq k$, the result is already proved in~\cite{guetta:poldcf}, so that
we may start the induction at level $k$. The case $n=k$ is already
covered by~\cite{guetta:poldcf}. To prove the induction step, we first
establish a few preliminary results.
\end{paragr}
\begin{paragr}\label{paragr:generators}
  Let $C$, $D$ be two $\pair{n{+}1}k$-categories and $f:C\to D$ a
  morphism. Suppose that $D$ is freely generated by a
  $\pair{n{+}1}k$-polygraph $Q$. Let $T=Q_{n+1}$ be a set of
  $(n{+}1)$-generators of $D$ and $q:T\to D_{n+1}$ the inclusion of
  generators. Now define $S\subset C_{n+1}$ by
  \[
 S=f_{n+1}^{-1}(q(T))=\setof{a\in C_{n+1}}{f_{n+1}(a)\in q(T)}.
\]
and $p:S\to C_{n+1}$ the inclusion map.
  Let $h:S\to T$ be the map taking $a\in S$ to the unique $b\in T$ such
  that $f_{n+1}p(a)=q(b)$. By definition, $f_{n+1}p=qh$. Therefore
  $C$, $D$, $f$, $p:S\to C_{n+1}$ and $q:T\to D_{n+1}$ satisfy the
  hypotheses of~\cref{lemma:main_diagram} and we get a commutative diagram of shape~(\ref{eq:main_diagram}). Moreover, as $D$ is freely
  generated by $Q$, the map $\fcls q:\wtxeq{T}\to D_{n+1}$ is a bijection.
\end{paragr}
\begin{lemma}\label{lemma:surj}
 If $f$ is a Conduché fibration, then the map $\fcls p:\wtxeq S\to
 C_{n+1}$ in diagram~(\ref{eq:main_diagram}) is surjective.
\end{lemma}
\begin{proof}
 We keep throughout the proof the notations of~\ref{paragr:addproperties} and~\cref{lemma:main_diagram}.
  By hypothesis, $\fcls q$ is bijective and we know that $\cans T$ is
surjective. Therefore $\extend q=\fcls q\cans T$ is surjective.  Thus,
for each $d\in D_{n+1}$, we choose a formula  $v_d\in \wtx T$ of
{\em minimal length} such
that $\extend q(v_d)=d$. Note that by construction $\scex
n^Tv_d=\sce n^D d$ and $\tgex n^Tv_d=\tge n^D d$.
We now show how to associate to each $c\in C_{n+1}$ a formula
$u_c\in\wtx S$ such that $\scex n^S u_c=\sce
n^C c$, $\tgex n^S u_c=\tge n^C c$ and $\extend p(u_c)=c$. The
correspondence $c\mapsto u_c$ is defined by structural induction on
the complexity of $v_d$ where $d=f_{n+1}(c)$:
\begin{itemize}
\item if $v_d=\fid y$ for some $y\in D_n$, then $d=\extend q(\fid
  y)=\unit{n+1}(y)$ and by the  Conduché condition~(1)
  there is a unique $x\in C_n$ such that $c=\unit{n+1}(x)$ and $f_n(x)=y$, in
  which case we choose  $u_c=\fid x$, whence $\scex n^Su_c=x=\sce n^Cc$
  and $\tgex n^Su_c=x=\tgex n^Cc$;   
\item if $v_d=\fcst b$ for $b\in T$, then $d=q(b)$ and as  $c\in
  f_{n+1}^{-1}(d)$, $c\in S$, so that $c=p(c)=\extend p(\fcst c)$ and we
  choose $u_c=\fcst c$, whence $\scex n^Su_c=\sce n^Cc$
  and $\tgex n^Su_c=\tgex n^Cc$; 
 \item if $v_d=\finv b$ for $b\in T$, then $f_{n+1}(\inv c)=\inv d$
   but $\inv d=\inv{\extend q(\finv b)}=\extend q(\fcst b)$, which
   implies $v_{\inv d}=\fcst b$ because $\fcst b$ is the unique
   formula of minimal length whose image by $\extend q$ is $\inv
   d$. Thus $\inv c\in S$ and we choose $u_c=\finv{\inv c}$, whence
   $\scex n^Su_c=\tge n^C\inv c=\sce n^Cc$ and
   $\tgex n^Su_c=\sce n^C\inv c=\tge n^Cc$;
 \item if $v_d$ is of the form $(v_1\fcomp i v_2)$ for $i\leq n$,
   $v_1,v_2\in \wtx T$, then $d_1=\extend q(v_1)$ and $d_2=\extend q
   (v_2)$ are $i$-composable and $d=d_1\comp i d_2$. By the
   Conduché condition~(2), there is a unique pair $c_1$, $c_2$ of
   $i$-composable cells in $C_{n+1}$ such that $f_{n+1}(c_1)=d_1$,
   $f_{n+1}(c_2)=d_2$ and $c=c_1\comp i c_2$. 
By the minimal lenght condition $\lgh
   v_{d_1}\leq \lgh v_1 $ and $\lgh v_{d_2}\leq \lgh v_2$, so that
   $\lgh v_{d_1}<\lgh v_d$ and $\lgh v_{d_2}<\lgh v_d$ and the
   induction hypothesis applies. Therefore we get
   $u_{c_1},u_{c_2}\in\wtx S$ such that $\extend p(u_{c_1})=c_1$ and
   $\extend p(u_{c_2})=c_2$. By induction hypothesis, the source and
   target of $u_{c_1}$ and $u_{c_2}$ are those of $c_1$ and $c_2$
   respectively. Therefore $\scex i^Su_{c_2}=\sce i^Cc_2=\tge
   i^Cc_1=\tgex i^Su_{c_1}$, so that the formula $(u_{c_1}\fcomp i
   u_{c_2})$ is well typed and we may define $u_c=(u_{c_1}\fcomp i
   u_{c_2})$. By definition, $\extend p(u_c)=c$. Finally we easily check
   that $\scex n^Su_c=\sce n^Cc$ and $\tgex n^Su_c=\tge n^C c$ in both
   cases $i=n$ and $i<n$. 
 \end{itemize}
 By constructing a section $c\mapsto u_c$ of the map $\extend p$ we
 have shown that $\extend p$ is surjective. As $\extend p=\fcls p\cans
 S$, this implies that $\fcls p$ is also surjective.
\end{proof}
\begin{paragr}\label{paragr:restricted_formulas}
  let $C$, $S$, $p:S\to C_{n+1}$ as in~\ref{paragr:generators}. For
  each $c\in C_{n+1}$ we define
  \[\wtxr cS=\setof{w\in\wtx S}{\extend p(w)=c}.\]
  Now, in the hypotheses of~\cref{lemma:main_diagram}, for each $c\in
  C_{n+1}$ and $d=f_{n+1}(c)$, if $w\in\wtxr cS$, then $\extend
  q\wtx\phi(w)=f_{n+1}\extend p(w)=f_{n+1}(c)=d$, so that
  $\wtx\phi(w)\in\wtxr dT$. We may therefore define
  \[\wtxr c\phi:\wtxr cS\to \wtxr dT\]
  as the restriction of $\wtx\phi$ to $\wtxr cS$.
\end{paragr}
\begin{lemma}\label{lemma:injectivity}
  If $f$ is a Conduché fibration, then for all $c\in C_{n+1}$, the map
  $\wtxr c\phi$ is injective.
\end{lemma}
\begin{proof}
  Let $c\in C_{n+1}$ and $g=\wtxr c\phi$. Note that for each $w\in\wtxr
  cS$, $\lgh g(w)=\lgh w$. Therefore it suffices to prove that for
  each $l>0$, the restriction of $g$ to the subset
  \[W_l=\setof{w\in\wtxr cS}{\lgh w=l}\]
  is injective. We reason by induction on $l>0$.
  \begin{enumerate}
  \item Suppose $l=1$ and let $w_1,w_2\in W_1$ such that
    $g(w_1)=g(w_2)$. Three cases are possible:
    \begin{enumerate}
    \item if $w_1=\fcst a$ for some $a\in S$, then
      $g(w_1)=\fcst{h(a)}=g(w_2)$ so that $w_2$ is of the form
      $\fcst{a'}$ for some $a'\in S$, whence
      $g(w_2)=\fcst{h(a')}=\fcst{h(a)}$. Now $p:S\to C_{n+1}$ is by
      definition an inclusion map, hence injective, and
      $p(a)=p(a')=c$. Therefore $a=a'$ and $w_1=w_2$;
    \item if $w_1=\finv{a}$ for some $a\in S$, then $w_2$ is of the
      form $\finv{a'}$ for some $a'\in S$. Therefore
      $\finv{h(a)}=g(w_1)=g(w_2)=\finv{h(a')}$, which implies
      $h(a)=h(a')$ and also
      $g(\fcst{a})=\fcst{h(a)}=\fcst{h(a')}=g(\fcst{a'})$. By the
      previous case, this implies $a=a'$ and $w_1=w_2$;
      \item if $w_1=\fid x$ for some $x\in C_n$ then
      $g(w_1)=\fid{f_n(x)}=g(w_2)$, whence $w_2$ is of the form
      $\fid{x'}$ for some $x'\in C_n$ such that $f_n(x')=f_n(x)$. Now
      $\unit{n+1}(x)=\extend p(w_1)=c=\extend p(w_2)=\unit{n+1}(x')$
      so that $x=x'$ and $w_1=w_2$.
    \end{enumerate}
    \item Suppose $l>0$ and let $w_1,w_2\in W_l$ such that
      $g(w_1)=g(w_2)$. Then $w_1$ is of the form $(w'_1\fcomp{i_1}
      w''_1)$ and  $w_2$ is of the form $(w'_2\fcomp{i_2} w''_2)$, but
      as $z=g(w_1)=g(w_2)$ is of the form $(z'\fcomp i z'')$,
      necessarily $i_1=i_2=i$,
      $g(w'_1)=g(w'_2)=z'$ and $g(w''_1)=g(w''_2)=z''$. Now
      $f_{n+1}(c)=\extend{q}(z')\comp i \extend{q}(z'')$ and by the
      Conduché condition~(2) there is a unique decomposition
      $c=c'\comp i c''$ such that $f_{n+1}(c')=\extend{q}(z')$ and
      $f_{n+1}(c'')=\extend{q}(z'')$. As $c=\extend{p}(w'_1)\comp
      i\extend{p}(w''_1)=\extend{p}(w'_2)\comp i\extend{p}(w''_2)$,
      $f_{n+1}\extend p(w'_1)=\extend qg(z')=\extend pg(w'_2)$ and
      $f_{n+1}\extend p(w''_1)=\extend qg(z'')=\extend pg(w''_2)$, the
      uniqueness of the decomposition implies $\extend p
      (w'_1)=\extend p(w'_2)=c'$ and  $\extend p(w''_1)=\extend p(w''_2)=c''$.
Therefore $w'_1,w'_2\in \wtxr {c'}S$ and $w''_1,w''_2\in \wtxr {c''}S$
so that the induction hypothesis applies, whence 
      $w'_1=w'_2$ and $w''_1=w''_2$, so that $w_1=w_2$.
  \end{enumerate}
\end{proof}
\begin{lemma}\label{lemma:lift_frel}
 Suppose $f$ is a Conduché fibration and let $c\in C_{n+1}$. For each $w,w'\in\wtxr c{S}$, if $\wtxr
  c\phi(w)\frel \wtxr c\phi (w')$, then $w\frel w'$.
\end{lemma}
\begin{proof}
  Let $g=\wtxr c\phi$ and $w,w'\in\wtxr cS$ such that $g(w)\frel
  g(w')$. Let us examine the different cases of~\ref{ssubsec:eqform}:
  In case~(1), there are two $i$-composable cells $y,y'\in D_n$ such that
    $g(w)=(\fid y\fcomp i \fid{y'})$ and $g(w')=\fid{y\comp i
      y'}$. Thus $w'$ is of the form $\fid z$ for some $z\in C_n$ such
    that $f_n(z)=y\comp i y'$. As $f$ is a Conduché fibration, there
    is a unique pair $x,x'\in C_n$ of $i$-composable cells such that
    $f_n(x)=y$, $f_n(x')=y'$ and $z=x'\comp i x'$, so that
    $w'=\fid{x\comp i x'}$. Now $w_1=(\fid x\fcomp i\fid {x'})$ belongs to $\wtxr cS$ and satisfies $g(w_1)=(\fid y\fcomp i\fid {y'})=g(w)$.
    By~\cref{lemma:injectivity}, $w=w_1=(\fid x\fcomp i
    \fid{x'})$. Therefore $w\frel w'$.
In the first part of case~(2)  there is $b:y\to y'\in T$ such that $g(w)=(\fid y\fcomp n
      \fcst b)$ and $w'=\fcst b$. Thus $w'$ is of the form $\fcst a$
      with $a:x\to x'\in S$, $b=h(a)$ and $y=f_n(x)$. Now $w_1=(\fid
      x\fcomp n \fcst a)$ belongs to $\wtxr cS$ and satisfies
      $g(w_1)=(\fid y\fcomp n\fcst
      a)=g(w)$. By~\cref{lemma:injectivity}, $w=w_1=(\fid x\fcomp n
      \fcst a)$. Therefore $w\frel w'$. The second part of case~(2) is
      similar.
   Cases~(3), (4) and (5) are treated as case~(2). Cases~(6) and (7)
   immediately follow from~\cref{lemma:injectivity} and the fact that
   $g$ preserves the structure of the formulas.   In the first part of
   case~(8), there is a $b:y\to y'\in T$ such that $g(w)=(\fcst b\comp
   n\finv b)$. Therefore $w$ is of the form $(\fcst{a_1}\comp
   n\finv{a_2})$ for some $a_1:x_1\to x'_1\in S$,  $a_1:x_2\to x'_2\in
   S$ such that $x_2=x'_1$, $h(a_1)=h(a_2)=b$ and $f_n(x_1)=f_n(x_2)=y$. Now
   $w_1=(\fcst{a_1}\fcomp n\finv{a_1})$ belongs to $\wtxr cS$ and
   $g(w_1)=(\fcst b\fcomp n\finv b)=g(w)$, so that,
   by~\cref{lemma:injectivity}, $w=w_1$. Also $g(w')=\fid
   y=g(\fid{x_1})$ so that, by~\cref{lemma:injectivity} again,
   $w'=\fid{x_1}$. Therefore $w\frel w'$ and we are done.
 \end{proof}
 \begin{lemma}\label{lemma:lift_cfrel}
 Suppose $f$ is a Conduché fibration and let $c\in C_{n+1}$. For each $w,w'\in\wtxr c{S}$, if $\wtxr
  c\phi(w)\cfrel \wtxr c\phi (w')$, then $w\cfrel w'$.
\end{lemma}
\begin{proof}
  Let $g=\wtxr c\phi$ and $w,w'\in\wtxr cS$ such that $g(w)\cfrel
  g(w')$. If (i)~$g(w)\frel g(w')$ we get $w\frel w'$
  by~\cref{lemma:lift_frel}, whence $w\cfrel w'$. If~(ii) $g(w)=(y\fcomp
  i v)$ and $g(w')=(y\fcomp i v')$ with $v\cfrel v'$, then $w$ is of
  the form $(x\fcomp i u)$ and $w'$ is of the form $(x'\fcomp i u')$
  with $g(x)=g(x')=v$,  $g(u)=v$ and
  $g(u')=v'$. Now $f_{n+1}(c)=\extend p(w)=\extend qg(w)=\extend q
  (y)\comp i\extend q (v)$. By the uniqueness property in Conduché
  condition~(2), there is a unique pair $c',c''\in C_{n+1}$ such that
  $c=c'\comp i c''$, $f_{n+1}(c')=\extend q(y)$ and
  $f_{n+1}(c'')=\extend q(v)$. Note also that $v\cfrel v'$ implies $v
  \freq v'$, hence $\extend q(v)=\extend q(v')$. On the other hand,
  $f_{n+1}(c)=f_{n+1}\extend p(x)\comp i f_{n+1}\extend p(u)=\extend q
  (y)\comp i\extend q(v)$, whence $\extend p(x)=c'$ and $\extend
  p(u)=c''$. The same argument applies to $w'$ and shows that $\extend
  p(x')=c'$ and $\extend p(u')=c''$. Therefore $x,x'\in\wtxr{c'}S$ and
  $u,u'\in\wtxr{c''}S$. Thus, 
by~\cref{lemma:injectivity}, $x=x'$ and by induction
$u\cfrel u'$, whence $w\cfrel w'$. The same argument applies to
case~(iii), where  $g(w)=(v\fcomp i y)$ and
  $g(w')=(v'\fcomp i y)$ with $v\cfrel v'$.
\end{proof}
