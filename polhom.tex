\section{Polygraphic homology}
\subsection{Abelianization of \ook k\nbd-categories}\label{subsec:abel}
\begin{paragr}
  Recall that we have a so-called ``abelianization'' functor $\abel
  \colon \oCat \to \Chp$ from the category of $\oo$\nbd-categories to
  the category of chain complexes in non-negative degree, defined in
  the following way. For an $\oo$\nbd-category $C$ and $n\geq 0$
  \[
    \abel_n(C):=\Z C_n/\sim_n,
  \]
  where $\Z C_n$ is the free abelian group on the set of $n$\nbd-cells
  of $C$ and $\sim_n$ is the smallest equivalence relation compatible with
  the abelian group structure such that $x\comp{i} y
  \simeq _n x + y$ for any $x,y\in C_n$ that are
  $\comp{i}$-composables. The boundary operator is defined as
  \[
    \begin{aligned}
      \partial \colon \abel_{n+1}(C) &\to \abel_{n}(C)\\
      [x] &\mapsto [\target x] - [\source x].
    \end{aligned}
  \]
  One checks that this is well-defined, that we have $\partial
  \circ \partial=0$ and that the correspondence $C \mapsto
  \abel(C)$ can be made functorial in an obvious
  way. Furthermore, $\abel$ admits a right adjoint. For
  details we refer to \cite{abgmmm:polybk}.

  By pre-composing with the inclusion functor, for any $k\geq 0$, we
  also get an abelinization functor
  \[
    \abelk{k} \colon \opCat{k} \overset{\oemb{k}}{\longrightarrow} \oCat \overset{\lambda}{\longrightarrow} \Chp.
  \]
  In particular, for $k=\oo$, we have $\abel=\abelk{\oo}$ by
  definition. Notice that since $\oemb{k}$ admits a right adjoint, so does $\abelk{k}$.
\end{paragr}

The abelianization of \ook k-polygraphs admits a useful explicit
description which we recall below. First, we need the following useful
lemma.
\begin{lemma}\label{lemma:delooping}
  Let $P$ be an $\ook k$\nbd-polygraph and $C=\frgp P$ the free $\ook
  k$\nbd-category on this polygraph. For any $n > 0$, and $G$ an
  abelian group, a function
  \[
    f \colon C_n \to G
  \]
  such that
  \begin{enumerate}
  \item $f(x \comp{k} y)=f(x)+f(y)$, for any $i$\nbd-composable pairs 
    $(x,y)$ of $n$\nbd-cells of $C$ with $i<n$,
  \item $f(\unit n(z))=0$, for any $(n-1)$\nbd-cell $z$,
  \end{enumerate}
  is completely and uniquely determined by its values $f(x)$ on the
  generators $x \in P_n$.
\end{lemma}
\begin{proof}
Notice first that for any $n>0$, any
abelian group $G$ defines an $n$\nbd-groupoid, denoted by $\B^nG$,
whose set of $i$\nbd-cell is a singleton for $i<n$ and $G$ for $i=n$,
and all composition operations are equal to the group
operation. Furthermore, for any $n$\nbd-category $C$, an
$n$\nbd-functor $C \to \B^nG$ amounts exactly to a function $f \colon
C_n \to G$ satisfying the conditions of the lemma. The conclusion
follows from the universal property of
\cref{prop:univprop}.
\end{proof}
\begin{paragr}
  In particular, for an $\ook k$\nbd-polygraph $P$ and $C=\frgp P$, for
  any $n > 0$, there exists a unique function
  \[
    f \colon C_n \to \Z P_n,
  \]
  such that
  \begin{enumerate}
  \item $f(x\comp{i}y)=f(x)+f(y)$, for $i<n$ and $(x,y)$ pair of
    $i$\nbd-composable $n$\nbd-cells,
  \item $f(\unit{n}(z))=0$ for any
    $(n-1)$\nbd-cell $z$,
  \item $f(x)=x$, for any $x \in P_n$.
  \end{enumerate}
  (Note that since $\Z P_n$ is free, condition (2) is
  actually implied by condition (1). This is because we have
  $\unit{n}(z)\comp{n-1}\unit{n}(z)=\unit{n}(z)$, hence
  $f(\unit{n}(z))=2f(\unit{n}(z))$.)
  We can now extend this function by linearity to a morphism of abelian
  group $\Z C_n \to \Z P_n$, which, by construction, is compatible
  with the quotient $\sim_n$ and thus defines a morphism of abelian
  group
  \[
    \wght \colon \abel_n(C) \to \Z P_n,
  \]
  such that $\wght ([x])=x$ for any $x \in P_n$.

  Furthermore, we consider the morphism of abelian group
  \[
    \iota \colon \Z P_n \to \abel_n(C),
  \]
  defined on $x \in P_n$ by $\iota(x)=[x]$
\end{paragr}
\begin{lemma}\label{lemma:abelpoly}
  The morphism of abelian group $\wght \colon \abel_n(C) \to \Z P_n$ is an
  isomorphism, with inverse $\iota \colon \Z P_n \to \abel_n(C)$.
\end{lemma}
\begin{proof}
  By construction, we have $\wght([x])=x$ for any $x \in P_n$, which
  proves that $\wght \circ \iota = \Unit{\Z P_n}$. For the other
  equality, it suffices to prove that for any $x \in C_n$, we have
  \[
    \iota(\wght[x])=[x].
  \]
  Notice that this equality is true for $x \in P_n$ by construction,
  and furthermore the functions $C_n \to \lambda_n(C)$, $x \mapsto
  \iota(\wght[x])$ and  $x
  \mapsto [x]$ satisfy the conditions of lemma
  \cref{lemma:delooping}. By the uniqueness part of this lemma, we
  deduce that these two functions are equal.
\end{proof}
We obtain now the explicit description of the abelianization of a free
$\ook k$\nbd-category on an $\ook k$\nbd-polygraph.
\begin{proposition}
  Let $k \geq 0$, $P$ an \ook k-polygraph and $C=\frgp P$ the free
  $\ook k$\nbd-category on this polygraph. Then $\abelk{k}(C)$
  is canonically isomorphic to the chain complex
  \[
    \Z P_0 \overset{\partial}{\longleftarrow} \Z P_1
    \overset{\partial}{\longleftarrow} \dots,
  \]
  where $\partial$ is defined on generator by
  \[
    \partial x = \wght([\target x]) - \wght([\source x]).
  \]
\end{proposition}
\begin{proof}
  Notice that $\abel_0(C)=\Z P_0$ by construction. The rest follows
  from \cref{lemma:abelpoly}.
\end{proof}
\begin{paragr}\label{paragr:abeldisk}
  As an immediate consequence of the previous proposition, in the case
  $k=\oo$, we can compute the abelianization of $\Dsk{n}$ for any $n \geq 0$.

  \[
    \abel_k(\Dsk{n})\simeq \begin{cases} \Z &\text{ if } k=n\\
      \Z\oplus
      \Z &\text{ if } 0 \leq k < n\end{cases}
  \]
  and $0$ elsewhere. For $n>0$ the differential $\partial
  \colon \abel_k(\Dsk{n}) \to \abel_{k-1}(\Dsk{n})$ sends $1$ to
  $(1,1)$ for $k=n$ and is given by the matrix $\begin{pmatrix} 1 & 1
    \\ -1 & -1 \end{pmatrix}$, 
  for $0 \leq k < n$.

  The abelianization of $\Sph{n}$ is computed similarly and is equal
  to that of $\Dsk{n}$, except for $\abel_n(\Sph{n})=0$. 
\end{paragr}
\begin{paragr}\label{paragr:abelloc}
  Let $0 \leq k < \omega$. We are now going to analyze the behaviour of
  the abelianization functor with respect to the left adjoint $\opifun
  k \colon
  \oCat \to \opCat{k}$ of the inclusion functor. By definition,
  we have a commutative triangle
  \[
    \begin{tikzcd}[column sep=small]
      \opCat{k}\ar[rr,"\oemb{k}"]\ar[rd,"\abelk{k}"']&& \oCat\ar[dl,"\abel"] \\
      &\Chp&.
  \end{tikzcd}
  \]
  Now, consider the
  following, \emph{a priori} not commutative, triangle
  \[
    \begin{tikzcd}[column sep=small]
      \oCat\ar[rr,"\opifun{k}"]\ar[rd,"\abel"']&& \opCat{k}\ar[dl,"\abelk{k}"] \\
    &\Chp&.
  \end{tikzcd}
\]
The unit defines a natural transformation
\[
 \abel\Longrightarrow \abel \oemb{k}\opifun{k}= \abelk{k}\opifun{k}
\]
\end{paragr}
\begin{lemma}\label{lemma:abelpi}
  The above natural transformation $\abel \Rightarrow \abelk{k}\opifun{k}
  $ is an isomorphism. 
\end{lemma}
\begin{proof}
  Consider $\mu \colon \Chp \to \oCat$ the right adjoint to the
  abelianization functor $\abel \colon \oCat \to \Chp$. A direct
  inspection of an explicit description of $\mu$ (see \cite[Paragraph
  17.4.3]{abgmmm:polybk}) shows that for any chain complex $K$,
  $\mu(K)$ is actually an \oo\nbd-groupoid. In particular, for any $k
  \geq 0$, the functor $\mu$ factors through $\opCat{k}$ as
  \[
    \Chp \overset{\mu_k}{\longrightarrow} \opCat{k}
    \overset{\oemb{k}}{\longrightarrow} \oCat.
  \]
  Note that since $\oemb{k}$ is fully faithful and injective on
  objects, such a $\mu_k$ is unique.
  As $\abel$ is left adjoint to
  $\mu$, and $\oemb{k}$ is fully faithful, it follows readily that
  $\abel \circ \oemb{k}=\abelk{k}$ is left adjoint to $\mu_k$ for
  any $k \geq 0$. By essential uniqueness of left adjoints, this
  proves that
  \[
    \abel\simeq \abelk{k}\opifun{k}.
  \]
  An easy verification left to the reader shows that this (iso)morphism
  is exactly the one constructed in \ref{paragr:abelloc}.
\end{proof}
\begin{paragr}\label{paragr:gray}
  Recall that the category $\oCat$ admits a monoidal category
  structure with the so-called ``Gray tensor product'',
  simply denoted by $\gray$, and with unit $\Dsk{0}$ (see for example
  \cite[Appendice A]{ara_maltsiniotis:joint}). More generally, for any $0 \leq k \leq \omega$, we have
  the induced tensor product $\grayk{k}$ on $\opCat{k}$ defined by the
  formula
  \[
    C \grayk{k} D :=\opifun{k}(\oemb{k}C\gray \oemb{k}D).
  \]
  Since the unit $\Dsk{0}$ is also the terminal object of $\opCat{k}$,
  for any \ook k\nbd-categories $C$ and $D$, there is a canonical
  morphism
  \[
    C\grayk{k} D \to C \times D,
  \]
  which is natural in $C$ and $D$. This is not an isomorphism, except
  if $C$ or $D$ is a $0$\nbd-category (that is, a set).

  
  Let $u, v \colon C \to D$ be two morphisms of $\opCat{k}$. We call
  \emph{oplax transformation} $\alpha \colon C \Rightarrow D$ a morphism
  \[
    \alpha \colon \opifun{k}(\Dsk{1})\grayk{k}C \to D
  \]
  such that the following diagram is commutative
  \[
    \begin{tikzcd}[column sep=large]
      \Sph{0}\grayk{k} C \ar[d] \ar[dr,"{\left(u,v\right)}"] & \\
      \opifun{k}(\Dsk{1})\grayk{k}C \ar[r,"\alpha"] & D. % \\
      % \Dsk{0}\grayk{k} C \ar[u]\ar[ur,"v"'] &
    \end{tikzcd}
  \]
  Here, we used the fact that $\opifun{k}(\Sph{0})=\Sph{0}$ as well as
  the canonical identification
  \[
    \begin{aligned}
      \Sph{0}\grayk{k}C &=(\Dsk{0}\coprod \Dsk{0})\grayk{k}C \\
      &\simeq \left(\Dsk{0}\grayk{k}C\right)\coprod
        \left(\Dsk{0}\grayk{k}C\right) \\
      &\simeq C \coprod C,
    \end{aligned}
\]
where the isomorphism on the second line comes from the fact that the
Gray tensor product is bi-closed.

If $f \colon D \to D'$ is another morphism of $\opCat{k}$, we define
an oplax transformation $f\wsk\alpha\colon fu \Rightarrow fv$ as the
following composition
\[
 \opifun{k}(\Dsk{1}) \grayk{k} C \overset{\alpha}{\longrightarrow} D
  \overset{f}{\longrightarrow} D'.
\]
Similarly, if $g \colon C' \to C$ is a morphism of $\opCat{k}$, we
define an oplax transformation $\alpha \wsk g \colon ug \Rightarrow
vg$ as the following composition
\[
  \begin{tikzcd}
    \opifun{k}(\Dsk{1})\grayk{k} C'
    \ar[rr,"\Unit{\opifun{k}(\Dsk{1})}\grayk{k}g"]&&
    \opifun{k}(\Dsk{1})\grayk{k} C\ar[r,"\alpha"]&
    D.
  \end{tikzcd}
\]

Finally, note that a \emph{natural transformation} $\gamma \colon u \Rightarrow
v$ amounts to a morphism
\[
  \gamma \colon \opifun{k}(\Dsk{1})\times C \to D
\]
such that the following diagram is commutative
  \[
    \begin{tikzcd}[column sep=large]
      \Sph{0}\times C \ar[d] \ar[dr,"{\left(u,v\right)}"] & \\
      \opifun{k}(\Dsk{1})\times C \ar[r,"\alpha"] & D. 
    \end{tikzcd}
  \]
  As before, we have used here the canonical identification
  $\Sph{0}\times C\simeq C \coprod C$. In particular, any such natural
  transformation defines an oplax transformation $u \Rightarrow v$,
  abusively still denoted by $\gamma$, defined as the composition
  \[
     \opifun{k}(\Dsk{1})\grayk{k}C \longrightarrow  \opifun{k}(\Dsk{1})\times C \overset{\gamma}{\longrightarrow} D.
  \]
\end{paragr}
\begin{proposition}\label{prop:abelmonoidal}
  For any $0 \leq k \leq \omega$, the abelianization functor
  \[
    \abelk{k} \colon \opCat{k} \to \Chp
  \]
  is strong monoidal with respect to the Gray tensor product $\grayk{k}$ on
  $\opCat{k}$ and the usual tensor product of chain complexes on $\Chp$.
\end{proposition}
\begin{proof}
  The case $k=\omega$ is standard (see for example \cite{}). The
  general case follows from  $\abelk{k}\simeq
  \abel\opifun{k}$ (\cref{lemma:abelpi}) and the fact
  that $\abel$ and $\opifun{k}$ are strong monoidal.
\end{proof}
\begin{corollary}\label{cor:abeltrans}
  For any $0 \leq k \leq \oo$, the abelianization functor
    \[
    \abelk{k} \colon \opCat{k} \to \Chp
  \]
  sends oplax transformations to chain homotopies. 
\end{corollary}
\begin{proof}
  Let
   \[
    \begin{tikzcd}[column sep=large]
      \Sph{0}\grayk{k} C \ar[d] \ar[dr,"{\left(u,v\right)}"] & \\
      \opifun{k}(\Dsk{1})\grayk{k}C \ar[r,"\alpha"'] & D.
    \end{tikzcd}
  \]
  be an oplax
  transformation. Using the computations from \ref{paragr:abeldisk},
  \cref{lemma:abelpi} and the strong monoidality of $\abelk{k}$
  (\cref{prop:abelmonoidal}), we get a commutative diagram
  \[
    \begin{tikzcd}[column sep=large]
     \abelk{k}(C)\oplus \abelk{k}(C) \ar[d] \ar[dr,"{\left(\abelk{k}(u),\abelk{k}(v)\right)}"] & \\
      \abel(\Dsk{1})\grayk{k}\abelk{k}(C) \ar[r,"\abelk{k}(\alpha)"'] & \abelk{k}(D).
    \end{tikzcd}
  \]
  By the description of $\abel(\Dsk{1})$ from \ref{paragr:abeldisk},
  this amounts exactly to a chain homotopy from $\abelk{k}(u)$ to $\abelk{k}(v)$.
\end{proof}
\begin{proposition}\label{prop:abelquillen}
  For any $k \geq 0$, the abelianization functor
  \[
    \abelk{k} \colon \opCat{k} \to \Chp
  \]
  is left Quillen, when $\npCat{\oo}{k}$ is equipped with the folk
  model structure and $\Chp$ with the projective model structure.
\end{proposition}
\begin{proof}
    The case $k=\omega$ is \cite[Theorem 22.2.7]{abgmmm:polybk}. For the
  general case, since the folk model structure on $\opCat{k}$ is right induced from
  the inclusion functor $\oemb{k} \colon \opCat{k} \to \oCat
  $, this means that a set of generating cofibrations (resp.\ trivial
  cofibrations) is given by the image by $\opifun{k} \colon \oCat
  \to \opCat{k}$ of generating cofibrations of
  $\opifun{k}$. The result follows then immediatly from \cref{lemma:abelpi}.
\end{proof}
\begin{paragr}\label{paragr:subtlety}
  The previous proposition contains an important subtlety.
  Even though we have a
  commutative triangle,
  \[
    \begin{tikzcd}[column sep=small]
      \opCat{k}\ar[rr,"\oemb{k}"]\ar[rd,"\abelk{k}"']&& \oCat\ar[dl,"\abel"] \\
      &\Chp&,
  \end{tikzcd}
\]
and even though $\oemb{k}$ passes to the homotopy categories, as
it trivially preserves the weak equivalences, this doesn't \emph{a
  priori} imply that the following triangle at the level of homotopy categories
\[
  \begin{tikzcd}[column sep=small]
    \Loc{\opCat{k}}\ar[rr,"\oemb{k}"]\ar[rd,"\LL \abelk{k}"']&&
    \Loc{\oCat}\ar[dl,"\LL \abel"] \\
    &\Loc{\Chp}&,
  \end{tikzcd}
\]
is also commutative, and the universal property of left derived functors
only yields a natural transformation
\begin{equation}\label{transnat}
  \LL \abel\circ \oemb{k} \Rightarrow \LL \abelk{k},
\end{equation}
which is \emph{a priori} not an isomorphism (and we do not know if it
is in general, see \cref{openquestion}.)

More concretly, this natural transformation can be defined as follows:
let $C$ be an \ook k\nbd-category and let
\[
  \frgp P\to C
\]
be a polygraphic resolution in $\opCat{k}$. Since $\oemb{k}$ does
not preserve cofibrant objects, in general we need to take a polygraphic
resolution in $\oCat$
\[
  \frcat Q \to \oemb{k}\frgp{P}.
\]
The above natural transformation is then
\[
  \LL \abel(\oemb{k}(C))=\abel(\frcat Q) \longrightarrow
  \abel(\oemb{k}(\frgp P))=\abelk{k}(\frgp P)=\LL \abelk{k}(C).
\]
\end{paragr}
\begin{definition}\label{def:hompol}
  Let $k\geq 0$. The \emph{$\pair \oo k$\nbd-polygraphic homology} of
  an \ook k\nbd-category $C$ is defined as
  \[
    \Hop{k}_{\bullet}(C):=\Ho_{\bullet}(\LL \abelk{k} (C)).
  \]
\end{definition}

\begin{paragr}\label{paragr:cancomp}
  Given an \ook k\nbd-category $C$, we shall
  simply write $\Hop{\oo}_{\bullet}(C)$ instead of
  $\Hop{\oo}_{\bullet}(\oemb{k}(C))$.
  In particular, the natural transformation \eqref{transnat} becomes
  \[
    \Hop{\oo}_{\bullet}(C) \to \Hop{k}_{\bullet}(C).
  \]
\end{paragr}

\begin{openquestion}\label{openquestion} With the same notation as
  above, is the canonical morphism
  \[
     \Hop{\oo}_{\bullet}(C) \to \Hop{k}_{\bullet}(C)
   \]
   an isomorphism of $\Loc{\Chp}$?
\end{openquestion}
As we shall see at the end of the next section, we answer positively to
the above question in a particular case.
\subsection{The two fundamental properties}
\begin{paragr}\label{paragr:colimtr}
  Let $C$ be a $1$\nbd-category and some $k \geq 1$. We define a functor
  \begin{align*}
    \tr{\opCat{k}}{C} &\to \fun{C}{\opCat{k}}\\
    (X, X \overset{p}{\to} C) &\mapsto \left(c \mapsto \tr{X}{c}\right).
  \end{align*}
  Notice that for $(X, X \overset{p}{\to} C)$ in $\tr{\opCat{k}}{C}$,
  we have a cocone
  \[
    (\tr{X}{c} \to X)_{c \in \Ob{C}},
  \]
  where each $\tr{X}{c} \to X$ is the canonical projection, and hence
  a canonical morphism
  \[
    \ilim_{c \in C}\tr{X}{c} \to X,
  \]
  which is natural in the sense that it defines a natural
  transformation from the functor
  \begin{align*}
    \tr{\opCat{k}}{C} &\to \opCat{k}\\
    (X, X \overset{p}{\to} C) &\mapsto \ilim_{c \in C} \tr{X}{c},
  \end{align*}
  to the projection functor
    \begin{align*}
    \tr{\opCat{k}}{C} &\to \opCat{k}\\
    (X, X \overset{p}{\to} C) &\mapsto X.
    \end{align*}
    Finally, if we take $C$ a groupoid and $k=0$, then everything said
    above still makes sense.
\end{paragr}
\begin{lemma}\label{lemma:colim}
  The natural morphism of $\opCat{k}$
  \[
     \ilim_{c \in C}\tr{X}{c} \to X,
   \]
   is an isomorphism.
\end{lemma}
\begin{proof}
The case $k=\oo$ is \cite[Lemma 7.5]{guetta:homcat}. The general case
follows from the fact that $\opCat{k} \to \oCat$
  preserves colimits.
\end{proof}
% Note that if we start with a groupoid $G$, then we proceed in exactly
% the same way
\begin{proposition}\label{prop:hocolimtr}
  Let $C$ be a $1$\nbd-category. Then for any $k\geq 1$, the canonical morphism of
  $\Loc{\opCat{k}}$
  \[
    \hoilim_{c \in C}\tr{C}{c} \to C
  \]
  is an isomorphism. If $C$ is a groupoid, then this also works for
  $k=0$.
\end{proposition}
\begin{proof}
  Let $\frgp{P} \to C$ be a polygraphic resolution of $C$ in
  $\opCat{k}$ and $\tr{\frgp P\!}{-} \colon C \to \opCat{k}$ obtained
  as in section \ref{subsec:polres}. We get a commutative diagram in
  $\Loc{\opCat{k}}$
  \[
    \begin{tikzcd}
      \displaystyle\hoilim_{c \in C}\tr{\frgp P\!}{c} \ar[d] \ar[r]& \displaystyle\ilim_{c \in
        C}\tr{\frgp P\!}{c} \ar[d] \ar[r] & \frgp P \ar[d] \\
      \displaystyle\hoilim_{c \in C}\tr{C}{c} \ar[r]&\displaystyle\ilim_{c \in C}\tr{C}{c}\ar[r] & C
    \end{tikzcd}
  \]
  Note that $\ilim_{c \in C}$ in the middle are the image by the
  localisation function \[\opCat{k} \to \Loc{\opCat{k}}\]
  of the actual
  colimits taken in the category $\opCat{k}$. The far right vertical
  arrow is an isomorphism by hypothesis, the horizontal arrows of the
  square on the right are isomorphism by \cref{lemma:colim}. The far left
  vertical arrow is an isomorphism because the natural transformation
  $\tr{\frgp P\!}{c} \to \tr{C}{c}$ is a pointwise weak equivalence by
  stability of trivial fibrations by pull back. Finally, the top
  horizontal arrow of the square on the left is an isomorphism
  because $\tr{\frgp P\!}{-}$ is cofibrant by \cref{thm:projresol} (or
  \cref{thm:projresol_grp} in the case $k=0$). By a two-out-of-three
  applied twice, we get that all the remaining arrows of this diagram
  are isomorphisms too. 
\end{proof}

\begin{lemma}\label{lemma:liftoplax}
  Let $0 \leq k \leq \omega$ and consider a diagram as below in $\opCat{k}$, 
  \[
    \begin{tikzcd}[row sep=large]
      A' \ar[d,"p"]\ar[r,bend left,"u'"] \ar[r,bend right,"v'"']& B' \ar[d,"q"] \\
      A \ar[r,bend left,"u",""{name=A,below}] \ar[r,bend
      right,"v"',""{name=B,above}]& B,
      \ar[from=A,to=B,Rightarrow,"\alpha"]
    \end{tikzcd}
  \]
  where the front and back faces are commutative. If $q$ is a trivial
  fibration and $A'$ is cofibrant, for the folk model structure on
  $\opCat{k}$, then there exists an oplax transformation $\alpha'
  \colon u' \Rightarrow v'$ making the whole diagram commutative, which means
  that
  \[
    q\wsk \alpha' = \alpha \wsk p
  \]
\end{lemma}
\begin{proof}
  The data of the given diagram amounts to the following commutative
  diagram in $\opCat{k}$
  \[
    \begin{tikzcd}
      \Sph{0} \grayk{k} A' \ar[r]\ar[d] \ar[rr,bend
      left,"{(u',v')}"]& \opifun{k}\Dsk{1} \grayk{k} A'
      \ar[d]& B' \ar[d,"q"] \\
      \Sph{0} \grayk{k} A \ar[r] & \opifun{k}\Dsk{1} \grayk{k} A \ar[r,"\alpha"]
      & B.
    \end{tikzcd}
  \]
  Since $A'$ is cofibrant and $\grayk{k}$ makes the folk model
  structure on $\opCat{k}$ a monoidal model structure \cite{ara_lucas:folmon}, the
  arrow $ \Sph{0}  \grayk{k} A' \to  \opifun{k}\Dsk{1} \grayk{k} A'$ is a
  cofibration. Since $q$ is a trivial fibration, there exists a
  morphism $\alpha' \colon \opifun{k}\Dsk{1}  \grayk{k} A' \to B'$ making the
  whole diagram commute.
\end{proof}
\begin{proposition}\label{prop:hlgycontrcat}
  Let $C$ be a small category with a terminal object, then for any $1
  \leq k \leq \oo$ the
  canonical morphism of $\Derp{\Z}$
  \[
    \Hop{k}(C) \to \Z,
  \]
  induced by the canonical functor $C \to \Term$, is an
  isomorphism. If $C$ is a groupoid, then it is also an isomorphism
  for $k=0$.
\end{proposition}
\begin{proof}
  Let us denote by $c_0$ the terminal object of $C$. We then have a
  natural transformation
    \[
    \begin{tikzcd}
      C \ar[r,bend left,"\Unit{C}",""{name=A,below}]
      \ar[r,bend right,"\cst{c_0}"',""{name=B,above}] & \C
      \ar[from=A,to=B,Rightarrow]
    \end{tikzcd}
  \]
  from the identity functor on $C$ to the constant functor with value
  $c_0$.  By \ref{paragr:gray}, this defines an oplax
  transformation. Now, let
  \[
    \frgp{P} \to C
  \]
  be a polygraphic resolution of $C$. Since trivial fibrations are
  surjective on objects, we can pick an object $x_0$ of $P$ whose image in
  $C$ is $c_0$. All in all, we have a diagram
  \[
    \begin{tikzcd}[row sep=huge]
      \frgp{P} \ar[d]\ar[r,bend left,"\Unit{\frgp{P}}"] \ar[r,bend
      right,"\cst{x_0}"']& \frgp{P} \ar[d] \\
      C \ar[r,bend left,"\Unit{C}",""{name=A,below}] \ar[r,bend
      right,"\cst{c_0}"',""{name=B,above}]& C,
      \ar[from=A,to=B,Rightarrow]
    \end{tikzcd}
  \]
  such that the front and back faces are commutative. Since
  $\frgp{P}$ is cofibrant, and since the vertical arrows are trivial
  fibrations, we can apply \cref{lemma:liftoplax}, which yields an
  oplax transformation from the identity \oo\nbd-functor on $P$ to
  the constant \oo\nbd-functor with value $x_0$.

  Finally, applying \cref{cor:abeltrans}, we get a chain homotopy
  between the identity morphism \[\abelk{k}(P) \to
    \abelk{k}(P)\] and the composition
  \[\abelk{k}(P) \to \Z \to
  \abelk{k}(P),\] which proves that  $\abelk{k}(P) \to
  \Z$ is a quasi-isomorphism.
\end{proof}

\begin{theorem}\label{thm:polhom}
  Let $C$ be a (1-)category. Then, for any $1
  \leq k \leq \oo$, there is an isomorphism
  \[
    \Hop{k}_{\bullet}(C) \to \Ho_{\bullet}(C)
  \]
  natural in $C$. If $C$ is a groupoid, this also holds
  for $k=0$. Furthermore, this
  isomorphism can be chosen natural in $k$ in the sense that for $k
  \leq k'$, the triangle 
  \[
    \begin{tikzcd}[column sep=small]
         \Hop{k}_{\bullet}(C)\ar[dr]
         \ar[rr]&&\Hop{k'}_{\bullet}(C)\ar[dl]\\
         &\Ho_{\bullet}(C)&
    \end{tikzcd}
  \]
  is commutative. 
\end{theorem}

\begin{proof}
  Let's start with the first part of the theorem. Recall that if $F
  \colon \M \to \M'$ is a functor, and $C$ is a small category we
  denote by $\fun{C}{F} \colon \fun{C}{\M} \to \fun{C}{\M'}$ the
  functor induced by post-composition. Consider the
  following zig-zag in $\Derp{\Z}$

  \[
    \begin{tikzcd}
      \hoilim_{C}p^*_C(\LL \abelk{k}(\Term))&&\\
    \hoilim_{C}\LL \fun{C}{\abelk{k}}(p_C^*\Term) \ar[u,"(a)"]&\ar[l,"(b)"'] \hoilim_{C}\fun{C}{\LL
     \abelk{k}}(\tr{C}{-})\ar[r,"(c)"]&  \LL
   \abelk{k}(\hoilim_{C}\tr{C}{-})\ar[d,"(d)"]\\
   && \LL\abelk{k}(C),
  \end{tikzcd}
  \]
  where
  \begin{itemize}
    \item $(a)$ and $(c)$ are the universal morphisms constructed in \ref{paragr:univmor},
    \item $(b)$ comes from the fact that $p_C^*\Term$ is the terminal
      object of $\fun{C}{\opCat{k}}$
      \item $(d)$ is induced by the cocone $(\tr{C}{c}\to
  C)_{c \in \Ob(C)}$ (see \ref{paragr:colimtr}).
\end{itemize}
All these morphisms of $\Derp{\Z}$ are in fact isomorphisms:
\begin{itemize}
  \item $(a)$ by \eqref{eq:morphder} of \cref{prop:htpycocts} and the fact that $\abelk{k}$
    is a left Quillen functor (\cref{prop:abelquillen}),
      \item $(b)$ by \cref{prop:der2}, \cref{prop:hlgycontrcat} and the
    fact that for each object $c$ of $C$, the category $\tr{C}{c}$ has
    a terminal object,
  \item $(c)$ by \eqref{eq:cocontinuous} of \cref{prop:htpycocts} and the fact that $\abelk{k}$
    is a left Quillen functor,
  \item $(d)$ by \cref{prop:hocolimtr}.
  \end{itemize}
  % Note that we have also implicitly used in the first two items the trivial fact that
  % $\hoilim_{C}$ preserves isomorphisms of $\Derp{C}$. And similarly
  % with $\LL \abelk{k}$ and isomorphisms of
  % $\Loc{\opCat{k}}$ for the third item.
  Finally, since by
  \cref{paragr:abeldisk} and the fact that $\Dsk{0}=\Term$, we have
  \[
    \LL \abelk{k}(\Term) \simeq \Z.
  \]
  All in all, this yields an isomorphism
  \[
    \Ho(C,\Z) \simeq \Hop{k}(C),
  \]
  which is natural in $C$ by construction, as all the morphisms in the
  previous zig-zag are.

  For the second part, consider the following diagram in $\Derp{\Z}$:
  \[
    \begin{tikzcd}
      \LL\abelk{k}(C)\ar[r] & \LL\abelk{k'}(C)\\
       \LL\abelk{k}(\hoilim_{C}\tr{C}{-})\ar[r]\ar[u]&
       \LL\abelk{k'}(\hoilim_{C}\tr{C}{-})\ar[u]\\
       \hoilim_{C}\fun{C}{\LL
     \abelk{k}}(\tr{C}{-})\ar[r] \ar[u]\ar[d]& \hoilim_{C}\fun{C}{\LL
     \abelk{k'}}(\tr{C}{-})\ar[u]\ar[d]\\
   \hoilim_{C}\LL \fun{C}{\abelk{k}}(p_C^*\Term)\ar[r] \ar[d]& \hoilim_{C}\LL
   \fun{C}{\abelk{k'}}(p_C^*\Term)\ar[d] \\
    \hoilim_{C}p^*_C(\LL \abelk{k}(\Term))\ar[r]& \hoilim_{C}p^*_C(\LL
    \abelk{k'}(\Term)),
    \ar[from=1-1,to=2-2,phantom,description, "(A)"]
    \ar[from=2-1,to=3-2,phantom,description,"(B)"]
    \ar[from=3-1,to=4-2,phantom,description,"(C)"]
    \ar[from=4-1,to=5-2,phantom,description,"(D)"]
  \end{tikzcd}
\]
where the horizontal arrows are induced by the natural transformation
\eqref{transnat} from \cref{paragr:subtlety}
\[
    \LL \abelk{k'}\circ \emb{k}{k'} \Rightarrow \LL \abelk{k}.
  \]
  \lgcomment{En fait, en \eqref{transnat} on parle seulement de
    $\oemb{k}$ et pas $\emb{k}{k'}$. Faut-il changer?}
  (Note that for simplicity we have not make appear the functor
  $\emb{k}{k'} \colon \Loc{\opCat{k}}\to \Loc{\opCat{k'}}$ in the above
  diagram.) This naturality implies that the squares $(A)$ and $(C)$ are
  commutative. Furthermore, the commutativity of the squares $(B)$ and
  $(D)$ follows from the naturality in $F$ of \eqref{eq:morphder} and
  \eqref{eq:cocontinuous} from \cref{prop:htpycocts}.

  To conclude, we only need to notice that the following triangle is
  commutative as an immediate computation shows
  \[
    \begin{tikzcd}
      \LL \abelk{k}(\Term) \ar[d,"\simeq"']\ar[r] & \LL \abelk{k'}(\Term)\ar[dl,"\simeq"]\\
      \Z.
    \end{tikzcd}
  \]
  
\end{proof}
